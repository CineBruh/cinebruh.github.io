/* =====================================================================
   Metas de donate (SOMENTE LEITURA, para embutir no Cinebruh)
   Lê a chave brunaSubathonGoals do SE_API.store e ouve kvstore:update.
   NUNCA grava no SE_API.store, NUNCA soma donates, NUNCA manda bônus pro
   cronômetro e NUNCA executa comandos do chat.
   ===================================================================== */
const KEY = 'brunaSubathonGoals';
const MAX_GOALS = 15;
const POLL_MS = 15000;

const CHECK = '<svg viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path d="M5 12.5l4.5 4.5L19 7.5" fill="none" stroke="currentColor" stroke-width="3.5" stroke-linecap="round" stroke-linejoin="round"/></svg>';

let F = {};
let goals = [];
let total = 0;
let ready = false;

const $ = id => document.getElementById(id);

function num(v, d) {
  const n = parseFloat(String(v).replace(',', '.'));
  return isFinite(n) ? n : d;
}

function parseAmount(s) {
  s = String(s).replace(/[^\d.,-]/g, '');
  if (s.includes(',')) s = s.replace(/\./g, '').replace(',', '.');
  else if (/^\d{1,3}(\.\d{3})+$/.test(s)) s = s.replace(/\./g, '');
  const n = parseFloat(s);
  return isFinite(n) ? n : NaN;
}

function readGoals() {
  const list = [];
  for (let i = 1; i <= MAX_GOALS; i++) {
    const amount = parseAmount(F['g' + i + 'Amount']);
    if (!isFinite(amount) || amount <= 0) continue;
    list.push({
      amount,
      title: String(F['g' + i + 'Title'] || '').trim(),
      hours: Math.max(0, num(F['g' + i + 'Hours'], 0))
    });
  }
  return list.sort((a, b) => a.amount - b.amount);
}

function bonusText(h) {
  const m = Math.round(h * 60);
  const hh = Math.floor(m / 60), mm = m % 60;
  return '+' + (hh ? hh + 'H' : '') + (mm ? (hh ? ' ' : '') + mm + 'MIN' : '');
}

function money(v) {
  return (F.currency || 'R$') + ' ' + v.toLocaleString('pt-BR', { minimumFractionDigits: v % 1 ? 2 : 0, maximumFractionDigits: 2 });
}

function applyStyle() {
  const r = document.documentElement.style;
  const set = (k, v) => { if (v !== undefined && v !== '') r.setProperty(k, v); };
  set('--accent', F.accentColor);
  set('--done', F.doneColor);
  set('--todo', F.todoColor);
  set('--labeltext', F.labelTextColor);
  set('--boxbg', F.boxColor);
  set('--border', F.borderColor);
  set('--text', F.textColor);
  set('--track', F.trackColor);
  set('--brdr', num(F.borderRadius, 23) + 'px');
  set('--labelSize', num(F.labelSize, 18) + 'px');
  set('--bigSize', num(F.bigSize, 30) + 'px');
  set('--goalSize', num(F.goalSize, 16) + 'px');
  set('--width', num(F.width, 380) + 'px');
  $('title').textContent = F.title || 'METAS';
  $('wrap').classList.toggle('rot', F.rotate !== false);
}

function doneCount() {
  return goals.filter(g => total + 1e-9 >= g.amount).length;
}

function buildList() {
  const list = $('list');
  list.innerHTML = '';
  goals.forEach(g => {
    const row = document.createElement('div');
    row.className = 'goal';
    const a = document.createElement('span');
    a.className = 'gamt';
    a.textContent = money(g.amount);
    const t = document.createElement('span');
    t.className = 'gtitle';
    t.textContent = g.title || '—';
    const c = document.createElement('span');
    c.className = 'gcheck';
    c.innerHTML = CHECK;
    row.append(a, t);
    if (g.hours > 0 && F.showBonus !== false) {
      const b = document.createElement('span');
      b.className = 'gbonus';
      b.textContent = bonusText(g.hours);
      row.appendChild(b);
    }
    row.appendChild(c);
    list.appendChild(row);
  });
}

function renderBar(done) {
  const bar = $('barbox');
  bar.style.display = F.showBar === false ? 'none' : '';

  const last = goals[goals.length - 1];
  const next = goals[done];
  const allDone = goals.length > 0 && done >= goals.length;
  const target = F.barMode === 'final' ? (last ? last.amount : 0) : (next || last || { amount: 0 }).amount;

  $('current').textContent = money(total);
  $('target').textContent = target ? '/ ' + money(target) : '';
  $('fill').style.width = (target ? Math.min(100, total / target * 100) : 0) + '%';

  let txt = '';
  if (F.showNext !== false) {
    if (allDone) txt = F.allDoneText || 'TODAS AS METAS BATIDAS!';
    else if (next) txt = (F.nextLabel || 'PRÓXIMA') + ': ' + (next.title || money(next.amount));
  }
  $('next').textContent = txt;
}

function renderList(done) {
  const view = $('listview');
  const rows = Array.from($('list').children);
  const hidden = F.showList === false || !rows.length;
  view.style.display = hidden ? 'none' : '';
  if (hidden) return;

  rows.forEach((r, i) => {
    r.classList.toggle('done', i < done);
    r.classList.toggle('current', i === done);
    r.classList.toggle('todo', i > done);
  });

  const vis = Math.max(1, Math.round(num(F.visibleGoals, 4)));
  const back = F.keepDoneVisible === false ? 0 : 1;
  const start = Math.max(0, Math.min(done - back, rows.length - vis));
  const end = Math.min(rows.length, start + vis) - 1;

  const top = rows[start].offsetTop;
  const bottom = rows[end].offsetTop + rows[end].offsetHeight;
  view.style.height = (bottom - top + 12) + 'px';
  $('list').style.transform = 'translateY(' + (-top) + 'px)';
}

function render() {
  if (!ready) return;
  const done = doneCount();
  renderBar(done);
  renderList(done);
}

function bump(el) {
  el.classList.remove('bump');
  void el.offsetWidth;
  el.classList.add('bump');
}

function confetti() {
  if (F.confetti === false) return;
  const layer = $('confetti');
  const W = window.innerWidth, H = window.innerHeight;
  const cols = [F.accentColor, F.doneColor, F.todoColor, '#f64c8f', '#8b3de6', '#fefefe'].filter(Boolean);
  for (let i = 0; i < 60; i++) {
    const p = document.createElement('i');
    const s = 6 + Math.random() * 7;
    p.style.left = Math.random() * W + 'px';
    p.style.top = '-20px';
    p.style.width = s + 'px';
    p.style.height = s * 1.5 + 'px';
    p.style.background = cols[i % cols.length];
    p.style.borderRadius = Math.random() < 0.4 ? '50%' : '2px';
    layer.appendChild(p);
    const dx = (Math.random() - 0.5) * 200;
    const rot = (Math.random() - 0.5) * 1080;
    p.animate(
      [{ transform: 'translate(0, 0) rotate(0deg)' }, { transform: 'translate(' + dx + 'px, ' + (H + 40) + 'px) rotate(' + rot + 'deg)' }],
      { duration: 2200 + Math.random() * 1600, delay: Math.random() * 400, easing: 'cubic-bezier(.2,.6,.4,1)', fill: 'forwards' }
    ).onfinish = () => p.remove();
  }
}

function applyExternalTotal(nextVal, animate) {
  if (!isFinite(nextVal) || nextVal < 0) return;
  const rounded = Math.round(nextVal * 100) / 100;
  if (Math.abs(rounded - total) < 1e-9) return;
  const before = doneCount();
  total = rounded;
  render();
  if (!animate) return;
  bump($('barbox'));
  const after = doneCount();
  if (after > before) {
    const rows = $('list').children;
    for (let i = before; i < after && i < rows.length; i++) {
      const r = rows[i];
      r.classList.add('justdone');
      setTimeout(() => r.classList.remove('justdone'), 700);
    }
    confetti();
  }
}

async function syncFromStore(animate) {
  try {
    const saved = await SE_API.store.get(KEY);
    if (saved && typeof saved === 'object' && isFinite(+saved.total)) {
      applyExternalTotal(+saved.total, animate);
    }
  } catch (e) {}
}

/* Ouve SOMENTE kvstore:update da chave KEY — nenhum evento de donate/chat é processado */
window.addEventListener('onEventReceived', obj => {
  if (!ready || !obj.detail || obj.detail.listener !== 'kvstore:update') return;
  const d = obj.detail.event && obj.detail.event.data;
  if (!d) return;
  const key = d.key || '';
  if (key !== KEY && key !== 'customWidget.' + KEY) return;
  const val = d.value;
  if (val && typeof val === 'object' && isFinite(+val.total)) {
    applyExternalTotal(+val.total, true);
  }
});

window.addEventListener('onWidgetLoad', async obj => {
  F = obj.detail.fieldData || {};
  applyStyle();
  goals = readGoals();
  buildList();
  ready = true;
  await syncFromStore(false);
  render();
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(render);
  setInterval(() => syncFromStore(true), POLL_MS);
});
